# -----------------------------------------------------------------------------
# DhanMITR Voice Module — STT / TTS pipeline
# -----------------------------------------------------------------------------
# Requires Python 3.11 or 3.12. Kokoro pins <3.13, so the repo's global 3.14
# interpreter will not work — use the module venv:
#
#   py -3.11 -m venv voice/.venv
#   voice/.venv/Scripts/activate          (Windows)
#   pip install -r voice/requirements.txt
#
# System dependencies:
#   ffmpeg    — decodes browser webm/opus into 16 kHz mono wav
#   espeak-ng — Kokoro's fallback grapheme-to-phoneme backend
#     Windows : winget install Gyan.FFmpeg ; winget install eSpeak-NG.eSpeak-NG
#     macOS   : brew install ffmpeg espeak-ng
#     Linux   : sudo apt install ffmpeg espeak-ng
#
# torch is CPU-only here. Install it first from the CPU index to avoid pulling
# ~2.5 GB of unused CUDA libraries:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
# -----------------------------------------------------------------------------
